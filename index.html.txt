    <script>
        const noBtn = document.getElementById('noBtn');
        const yesBtn = document.getElementById('yesBtn');
        
        // Position the No button right next to the Yes button initially
        function initButtonPosition() {
            const yesRect = yesBtn.getBoundingClientRect();
            noBtn.style.left = (yesRect.right + 20) + 'px';
            noBtn.style.top = yesRect.top + 'px';
        }
        
        // Run initial positioning after page loads
        window.addEventListener('DOMContentLoaded', initButtonPosition);
        window.addEventListener('resize', initButtonPosition);

        // Track mouse movement across the entire screen
        document.addEventListener('mousemove', (e) => {
            const mouseX = e.clientX;
            const mouseY = e.clientY;

            // Get the center position of the No button
            const btnRect = noBtn.getBoundingClientRect();
            const btnCenterX = btnRect.left + btnRect.width / 2;
            const btnCenterY = btnRect.top + btnRect.height / 2;

            // Calculate the distance between the mouse and the button
            const distanceX = mouseX - btnCenterX;
            const distanceY = mouseY - btnCenterY;
            const distance = Math.sqrt(distanceX * distanceX + distanceY * distanceY);

            // Radius of sensitivity: how close the mouse gets before the button runs (in pixels)
            const sensitivityRadius = 100; 

            if (distance < sensitivityRadius) {
                // Calculate direction to run away (opposite of mouse direction)
                const dirX = distance === 0 ? 1 : -distanceX / distance;
                const dirY = distance === 0 ? 1 : -distanceY / distance;

                // Speed factor of how fast it pushes away
                const pushSpeed = 15; 
                
                let newX = btnRect.left + dirX * pushSpeed;
                let newY = btnRect.top + dirY * pushSpeed;

                // --- Screen Boundaries (Keeps it on the screen completely) ---
                const padding = 20; 
                const minX = padding;
                const minY = padding;
                const maxX = window.innerWidth - btnRect.width - padding;
                const maxY = window.innerHeight - btnRect.height - padding;

                // Stop at boundaries so it stays visible
                if (newX < minX) newX = minX;
                if (newX > maxX) newX = maxX;
                if (newY < minY) newY = minY;
                if (newY > maxY) newY = maxY;

                // Apply the new coordinates
                noBtn.style.left = newX + 'px';
                noBtn.style.top = newY + 'px';
            }
        });

        // Mobile touch support: jump away instantly if tapped
        noBtn.addEventListener('touchstart', (e) => {
            e.preventDefault(); 
            
            const padding = 40;
            const maxX = window.innerWidth - noBtn.offsetWidth - padding;
            const maxY = window.innerHeight - noBtn.offsetHeight - padding;
            
            noBtn.style.left = Math.max(padding, Math.floor(Math.random() * maxX)) + 'px';
            noBtn.style.top = Math.max(padding, Math.floor(Math.random() * maxY)) + 'px';
        });

        // Function to handle the "Yes" click
        function submitYes() {
            const form = document.getElementById('loveForm');
            const container = document.getElementById('mainCard');
            
            if(form.action.includes("YOUR_FORMSPREE_ID")) {
                alert("🎉 She said YES! (Remember to add your Formspree ID to the code to get email alerts!)");
                showSuccessPage(container);
            } else {
                fetch(form.action, {
                    method: 'POST',
                    body: new FormData(form),
                    headers: { 'Accept': 'application/json' }
                }).then(() => {
                    showSuccessPage(container);
                }).catch(() => {
                    showSuccessPage(container);
                });
            }
        }

        function showSuccessPage(container) {
            container.innerHTML = `
                <div class="gif-container">
                    <img src="https://media.giphy.com/media/v1.Y2lkPTc5MGI3NjExbW95M3N6cXN6bXN6cXN6bXN6cXN6bXN6cXN6bXN6cXN6ZSZlcD12MV9pbnRlcm5hbF9naWZfYnlfaWQmY3Q9cw/26vUxO3A8WVYqXwX6/giphy.gif" alt="Happy dance">
                </div>
                <h1>Yayyy! 🎉🥳</h1>
                <p>I knew you couldn't resist! I'll text you to figure out the details. See you soon! ❤️</p>
            `;
            // Hide the floating runaway button once she says yes
            noBtn.style.display = 'none';
        }
